    // RENDER METRICS & OVERVIEW
    function renderMetrics() {
      const total = trades.length;
      document.getElementById('statTotalTrades').innerText = total;

      if (total === 0) {
        document.getElementById('statNetR').innerText = '0.0R';
        document.getElementById('statAvgR').innerText = '0.0R';
        document.getElementById('statWinRate').innerText = '0%';
        document.getElementById('statWinCount').innerText = '0W';
        document.getElementById('statLossCount').innerText = '0L';
        document.getElementById('statBECount').innerText = '0BE';
        document.getElementById('statTopSetup').innerText = '-';
        renderChart([]);
        return;
      }

      let netR = 0;
      let wins = 0;
      let losses = 0;
      let bes = 0;
      let setupRMap = {};

      trades.forEach(t => {
        const r = parseFloat(t.rMultiple) || 0;
        netR += r;
        if (t.result === 'WIN') wins++;
        else if (t.result === 'LOSS') losses++;
        else bes++;

        // Setup performance
        setupRMap[t.setup] = (setupRMap[t.setup] || 0) + r;
      });

      const avgR = (netR / total).toFixed(2);
      const winRate = ((wins / total) * 100).toFixed(1);

      // Best setup
      let bestSetup = '-';
      let maxSetupR = -999;
      for (const [setup, rVal] of Object.entries(setupRMap)) {
        if (rVal > maxSetupR) {
          maxSetupR = rVal;
          bestSetup = setup;
        }
      }

      const netREl = document.getElementById('statNetR');
      netREl.innerText = (netR >= 0 ? '+' : '') + netR.toFixed(1) + 'R';
      netREl.className = `text-2xl sm:text-3xl font-bold mono ${netR >= 0 ? 'text-emerald-400' : 'text-red-400'}`;

      document.getElementById('statAvgR').innerText = (avgR >= 0 ? '+' : '') + avgR + 'R';
      document.getElementById('statWinRate').innerText = winRate + '%';
      document.getElementById('statWinCount').innerText = wins + 'W';
      document.getElementById('statLossCount').innerText = losses + 'L';
      document.getElementById('statBECount').innerText = bes + 'BE';
      document.getElementById('statTopSetup').innerText = bestSetup;

      // Prepare equity data chronologically
      const sortedByDate = [...trades].sort((a,b) => new Date(a.date) - new Date(b.date));
      let cumR = 0;
      const chartPoints = sortedByDate.map(t => {
        cumR += (parseFloat(t.rMultiple) || 0);
        return { date: t.date, r: parseFloat(cumR.toFixed(2)) };
      });

      renderChart(chartPoints);
    }

    // CHART.JS MINI EQUITY CURVE
    function renderChart(points) {
      const ctx = document.getElementById('equityChart').getContext('2d');
      if (equityChart) {
        equityChart.destroy();
      }

      const labels = points.map((p, i) => `#${i+1}`);
      const data = points.map(p => p.r);
      const isPositive = data.length > 0 && data[data.length - 1] >= 0;
      const color = isPositive ? '#10b981' : '#ef4444';

      equityChart = new Chart(ctx, {
        type: 'line',
        data: {
          labels: labels.length ? labels : ['0'],
          datasets: [{
            data: data.length ? data : [0],
            borderColor: color,
            borderWidth: 2,
            tension: 0.35,
            fill: true,
            backgroundColor: isPositive ? 'rgba(16, 185, 129, 0.08)' : 'rgba(239, 68, 68, 0.08)',
            pointRadius: data.length > 15 ? 0 : 3,
            pointHoverRadius: 5,
            pointBackgroundColor: color
          }]
        },
        options: {
          responsive: true,
          maintainAspectRatio: false,
          plugins: {
            legend: { display: false },
            tooltip: {
              callbacks: {
                label: (ctx) => `Kümülatif: ${ctx.parsed.y}R`
              }
            }
          },
          scales: {
            x: { display: false },
            y: {
              grid: { color: 'rgba(255,255,255,0.05)' },
              ticks: { color: '#64748b', font: { size: 10 } }
            }
          }
        }
      });
    }

    // RENDER TRADE CARDS WITH FILTERS & SORT
    function renderAll() {
      renderMetrics();

      const search = searchInput.value.toLowerCase().trim();
      const resFilter = filterResult.value;
      const dirFilter = filterDirection.value;
      const setupFilter = filterSetup.value;
      const sort = sortBy.value;

      let filtered = trades.filter(t => {
        const matchesSearch = 
          t.pair.toLowerCase().includes(search) ||
          t.setup.toLowerCase().includes(search) ||
          (t.thesis && t.thesis.toLowerCase().includes(search)) ||
          (t.lessons && t.lessons.toLowerCase().includes(search));

        const matchesRes = resFilter === 'ALL' || t.result === resFilter;
        const matchesDir = dirFilter === 'ALL' || t.direction === dirFilter;
        const matchesSetup = setupFilter === 'ALL' || t.setup === setupFilter;

        return matchesSearch && matchesRes && matchesDir && matchesSetup;
      });

      // Sorting
      filtered.sort((a, b) => {
        if (sort === 'date-desc') return new Date(b.date) - new Date(a.date);
        if (sort === 'date-asc') return new Date(a.date) - new Date(b.date);
        if (sort === 'r-desc') return (parseFloat(b.rMultiple) || 0) - (parseFloat(a.rMultiple) || 0);
        if (sort === 'r-asc') return (parseFloat(a.rMultiple) || 0) - (parseFloat(b.rMultiple) || 0);
        return 0;
      });

      tradesGrid.innerHTML = '';

      if (filtered.length === 0) {
        tradesGrid.classList.add('hidden');
        emptyState.classList.remove('hidden');
        return;
      }

      tradesGrid.classList.remove('hidden');
      emptyState.classList.add('hidden');

      filtered.forEach(trade => {
        const card = createTradeCard(trade);
        tradesGrid.appendChild(card);
      });
    }

    // EVENT LISTENERS SETUP
    function setupEventListeners() {
      // Modal controls
      document.getElementById('openModalBtn').addEventListener('click', () => openTradeModal());
      document.getElementById('closeModalBtn').addEventListener('click', () => tradeModal.classList.add('hidden'));
      document.getElementById('cancelModalBtn').addEventListener('click', () => tradeModal.classList.add('hidden'));

      document.getElementById('closeDetailBtn').addEventListener('click', () => detailModal.classList.add('hidden'));

      // Filter events
      searchInput.addEventListener('input', renderAll);
      filterResult.addEventListener('change', renderAll);
      filterDirection.addEventListener('change', renderAll);
      filterSetup.addEventListener('change', renderAll);
      sortBy.addEventListener('change', renderAll);

      // Auto set +/- sign on R depending on result
      document.getElementById('tradeResult').addEventListener('change', (e) => {
        const rInput = document.getElementById('tradeR');
        let current = Math.abs(parseFloat(rInput.value) || 0);
        if (e.target.value === 'LOSS') {
          rInput.value = current === 0 ? -1 : -current;
        } else if (e.target.value === 'WIN') {
          rInput.value = current === 0 ? 2 : current;
        } else {
          rInput.value = 0;
        }
      });

      // Dropzone and Image upload
      dropZone.addEventListener('click', () => imageFileInput.click());

      dropZone.addEventListener('dragover', (e) => {
        e.preventDefault();
        dropZone.classList.add('border-emerald-500');
      });
      dropZone.addEventListener('dragleave', () => dropZone.classList.remove('border-emerald-500'));
      dropZone.addEventListener('drop', (e) => {
        e.preventDefault();
        dropZone.classList.remove('border-emerald-500');
        if (e.dataTransfer.files && e.dataTransfer.files[0]) {
          handleFile(e.dataTransfer.files[0]);
        }
      });

      imageFileInput.addEventListener('change', (e) => {
        if (e.target.files && e.target.files[0]) {
          handleFile(e.target.files[0]);
        }
      });

      imageUrlInput.addEventListener('input', (e) => {
        const url = e.target.value.trim();
        if (url) {
          imagePreview.src = url;
          imagePreview.classList.remove('hidden');
          previewPlaceholder.classList.add('hidden');
          removeImgBtn.classList.remove('hidden');
        }
      });

      removeImgBtn.addEventListener('click', () => {
        imagePreview.src = '';
        imageFileInput.value = '';
        imageUrlInput.value = '';
        imagePreview.classList.add('hidden');
        previewPlaceholder.classList.remove('hidden');
        removeImgBtn.classList.add('hidden');
      });

      // Form Submit
      tradeForm.addEventListener('submit', (e) => {
        e.preventDefault();
        const editId = document.getElementById('editTradeId').value;

        const tradeData = {
          id: editId || 'trade_' + Date.now(),
          pair: document.getElementById('tradePair').value.toUpperCase().trim(),
          direction: document.getElementById('tradeDirection').value,
          result: document.getElementById('tradeResult').value,
          rMultiple: parseFloat(document.getElementById('tradeR').value) || 0,
          setup: document.getElementById('tradeSetup').value,
          timeframe: document.getElementById('tradeTimeframe').value,
          date: document.getElementById('tradeDate').value,
          entry: document.getElementById('tradeEntry').value.trim(),
          sl: document.getElementById('tradeSL').value.trim(),
          tp: document.getElementById('tradeTP').value.trim(),
          image: imagePreview.src || '',
          thesis: document.getElementById('tradeThesis').value.trim(),
          lessons: document.getElementById('tradeLessons').value.trim()
        };

        if (editId) {
          const index = trades.findIndex(t => t.id === editId);
          if (index !== -1) trades[index] = tradeData;
        } else {
          trades.unshift(tradeData);
        }

        saveTrades();
        renderAll();
        tradeModal.classList.add('hidden');
      });

      // Detail Actions (Delete / Edit)
      document.getElementById('deleteTradeBtn').addEventListener('click', () => {
        if (!selectedTradeId) return;
        trades = trades.filter(t => t.id !== selectedTradeId);
        saveTrades();
        renderAll();
        detailModal.classList.add('hidden');
      });

      document.getElementById('editTradeBtn').addEventListener('click', () => {
        detailModal.classList.add('hidden');
        openTradeModal(selectedTradeId);
      });

      // Export JSON backup
      document.getElementById('exportBtn').addEventListener('click', () => {
        const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(trades, null, 2));
        const dlAnchorElem = document.createElement('a');
        dlAnchorElem.setAttribute("href", dataStr);
        dlAnchorElem.setAttribute("download", `trading_journal_${new Date().toISOString().split('T')[0]}.json`);
        dlAnchorElem.click();
      });
    }

    // Convert file to Base64 Data URL for local storage
    function handleFile(file) {
      if (!file.type.startsWith('image/')) return;
      const reader = new FileReader();
      reader.onload = function(e) {
        imagePreview.src = e.target.result;
        imagePreview.classList.remove('hidden');
        previewPlaceholder.classList.add('hidden');
        removeImgBtn.classList.remove('hidden');
        imageUrlInput.value = ''; // clear url input if file uploaded
      };
      reader.readAsDataURL(file);
    }

    // Start App
    window.addEventListener('DOMContentLoaded', init);
  </script>
